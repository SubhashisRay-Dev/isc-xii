Sample Paper 3 · Question 3

TicketDateAnalyzer

Question

An airline records three dates for a passenger: Booking, Travel and Return. Each date is entered as day, month, year. Assume all dates are valid, in the same year, and the year is between 2024 and 2030. Design a class TicketDateAnalyzer:

  1. Find and display the day number of the year for each date (15 February = Day 46).
  2. Travel date must be after booking, else display “Invalid Booking” and stop. Return date must be after travel, else “Invalid Return” and stop.
  3. Booking Gap = days between booking and travel. Trip Duration = days between travel and return.
  4. If Booking Gap is at least twice the Trip Duration: “Advance Saver”. If Trip Duration is at least twice the Booking Gap: “Long-Stay Premium”. Otherwise “Standard Fare”.

Example 1: 01 01 2026, 25 01 2026, 05 02 2026 → Day numbers 1, 25, 36; Booking Gap 24; Trip Duration 11; Advance Saver.

Algorithm

  1. Start.
  2. Accept day, month and year of the booking, travel and return dates.
  3. For each date, compute the day number: add the days of all complete months before it (February has 29 days in a leap year) and the day of the month. Print the three day numbers.
  4. If tv <= bk print Invalid Booking and stop.
  5. If rt <= tv print Invalid Return and stop.
  6. gap = tv - bk; trip = rt - tv. Print both.
  7. If gap >= 2*trip print Advance Saver; else if trip >= 2*gap print Long-Stay Premium; else Standard Fare.
  8. Stop.

Variable description

VariableData typeDescription
d[], m[], y[]intDay, month and year of the three dates (0 booking, 1 travel, 2 return)
dm[]intDays in each month (February adjusted for leap year)
bk, tv, rtintDay numbers of booking, travel and return
gapintBooking gap in days
tripintTrip duration in days
nintRunning day count in dayNo()

Program (Java)

import java.util.*;

public class TicketDateAnalyzer {
    int d[] = new int[3], m[] = new int[3], y[] = new int[3];

    boolean isLeap(int yr) {                 // leap year rule
        return (yr % 4 == 0 && yr % 100 != 0) || (yr % 400 == 0);
    }

    // day number of the year for date dd/mm/yy
    int dayNo(int dd, int mm, int yy) {
        int dm[] = {31, 28, 31, 30, 31, 30, 31, 31, 30, 31, 30, 31};
        if (isLeap(yy)) dm[1] = 29;
        int n = dd;
        for (int i = 0; i < mm - 1; i++)     // add complete months before mm
            n += dm[i];
        return n;
    }

    void accept() {
        Scanner sc = new Scanner(System.in);
        String name[] = {"Booking", "Travel", "Return"};
        for (int i = 0; i < 3; i++) {
            System.out.print(name[i] + " Date (day month year): ");
            d[i] = sc.nextInt();
            m[i] = sc.nextInt();
            y[i] = sc.nextInt();
        }
    }

    void analyse() {
        int bk = dayNo(d[0], m[0], y[0]);
        int tv = dayNo(d[1], m[1], y[1]);
        int rt = dayNo(d[2], m[2], y[2]);
        System.out.println("Booking Day Number: " + bk);
        System.out.println("Travel Day Number: " + tv);
        System.out.println("Return Day Number: " + rt);

        if (tv <= bk) {                      // dates are in the same year, so day numbers can be compared
            System.out.println("INVALID BOOKING");
            return;
        }
        if (rt <= tv) {
            System.out.println("INVALID RETURN");
            return;
        }

        int gap = tv - bk;
        int trip = rt - tv;
        System.out.println("Booking Gap: " + gap + " days");
        System.out.println("Trip Duration: " + trip + " days");

        if (gap >= 2 * trip)      System.out.println("Fare Type: Advance Saver");
        else if (trip >= 2 * gap) System.out.println("Fare Type: Long-Stay Premium");
        else                      System.out.println("Fare Type: Standard Fare");
    }

    public static void main(String[] args) {
        TicketDateAnalyzer ob = new TicketDateAnalyzer();
        ob.accept();
        ob.analyse();
    }
}
